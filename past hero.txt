    js 

     <script>
      // =========================
      // MAIN SWIPER
      // =========================

      const mainSwiper = new Swiper(".hero-main", {
        effect: "fade",

        fadeEffect: {
          crossFade: true,
        },

        speed: 900,

        autoplay: {
          delay: 6000,
          disableOnInteraction: false,
        },
      });

      // =========================
      // CUSTOM VERTICAL PAGINATION
      // =========================

      const paginationItems = document.querySelectorAll(".pagination-item");

      // Update active pagination item

      function updatePagination(index) {
        paginationItems.forEach((item, i) => {
          item.classList.toggle("active", i === index);
        });
      }

      // Pagination click

      paginationItems.forEach((item) => {
        item.addEventListener("click", () => {
          const index = Number(item.dataset.slide);

          mainSwiper.slideTo(index);
        });
      });

      // Update pagination when slide changes

      mainSwiper.on("slideChange", () => {
        updatePagination(mainSwiper.realIndex);
      });
    </script>
